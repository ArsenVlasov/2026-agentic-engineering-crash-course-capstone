import { baseAdminPanelExtendedTest as test } from '@/tests/specs/regression/admin-panel/base-admin-panel-extended';
import { AdminSidebarTab } from '@/utils/constants/admin/admin-sidebar-navigation';
import { AUTOTEST_ALLURE, AUTOTEST_PLAYWRIGHT_TAGS } from '@/utils/constants/autotest-labels';
import { expect } from '@playwright/test';
import { linkTestCase } from '@/utils/helpers/allure-helper';
import { labelSuite } from '@/utils/helpers/label-suite';
import {
  readVisibleMentorNames,
  shouldSeeMultipleMentors,
  shouldSeeOnlyMentor,
} from '@/tests/specs/regression/admin-panel/mentor-stats/helpers/mentor-stats-filters';

test.use({ adminUserKey: 'user_TC_84_admin' });

test.beforeEach(async () => {
  await labelSuite({ feature: AUTOTEST_ALLURE.feature.mentors });
});

test.describe(
  'Верифікація фільтрів в таблиці менторів',
  {
    tag: [AUTOTEST_PLAYWRIGHT_TAGS.adminPanel, AUTOTEST_PLAYWRIGHT_TAGS.capability.mentors],
  },
  () => {
    test.beforeAll(() => linkTestCase(84));

    test('PRJ-TC-84 Фільтр «Ментор»: вибір і очищення', async ({
      adminUser,
      adminLoginPage,
      adminHomePage,
      mentorStatsPage,
    }) => {
      await test.step('1. Відкрити таблицю статистики менторів і побачити фільтри', async () => {
        await adminLoginPage.authenticate(adminUser);
        await adminHomePage.shouldSeePage();
        await adminHomePage.sidebar.selectSidebarTab(AdminSidebarTab.RESOURCES.MENTOR_STATS);
        await mentorStatsPage.shouldSeePage();
        await mentorStatsPage.filters.shouldSeeMentorAndPageSizeFilters();
      });

      const preferredNames = await readVisibleMentorNames(mentorStatsPage);

      await test.step('2. Вибрати ментора і застосувати фільтр', async () => {
        const mentorName = await mentorStatsPage.filters.selectListedMentor(preferredNames);
        await mentorStatsPage.filters.clickApply();
        await mentorStatsPage.table.waitForTableLoad();
        await adminLoginPage.page.waitForTimeout(3000);
        expect(await mentorStatsPage.table.getRowsCount()).toBeGreaterThan(0);
      });

    });
  },
);
